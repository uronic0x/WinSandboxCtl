cmake_minimum_required(VERSION 3.16)
project(WinSandboxCtl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8 /EHsc)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
add_executable(SandboxHost src/main.cpp)
add_executable(core-tests tests/core_tests.cpp)
target_link_libraries(SandboxHost PRIVATE Threads::Threads)
target_link_libraries(core-tests PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(SandboxHost PRIVATE shell32 advapi32 ole32 ws2_32)
  target_link_libraries(core-tests PRIVATE ws2_32 shell32)
  add_executable(DemoProbe tests/DemoProbe.cpp)
  target_link_libraries(DemoProbe PRIVATE advapi32)
endif()
# Preserve a self-contained executable + guest script layout for every generator.
add_custom_command(TARGET SandboxHost POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:SandboxHost>/guest"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/guest/guest.ps1" "$<TARGET_FILE_DIR:SandboxHost>/guest/guest.ps1")
enable_testing()
add_test(NAME core COMMAND core-tests)
